<html lang="zh-CN">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <script src="https://unpkg.com/vue@3"></script>
    <title>昼夜模式切换</title>
    <style>
        * {
            margin: 0;
            padding: 0;
            box-sizing: border-box;
        }
        /* 定义CSS变量统一管理颜色 */
        .theme-light {
            --bg-color: #ffffff;
            --text-color: #000000;
            --divider-color: #000000;
        }
        .theme-dark {
            --bg-color: #000000;
            --text-color: #f0ffff;
            --divider-color: #f0ffff;
        }
        .theme-container {
            min-height: 100vh;
            background-color: var(--bg-color);
            padding: 20px;
            transition: all 0.3s ease;
        }
        .theme-title {
            font-size: large;
            font-weight: bolder;
            color: var(--text-color);
            margin-bottom: 10px;
        }
        .theme-divider {
            border: none;
            background-color: var(--divider-color);
            height: 2px;
            margin-bottom: 10px;
        }
        .theme-text {
            color: var(--text-color);
            margin-bottom: 20px;
        }
        .theme-switch label {
            color: var(--text-color);
            cursor: pointer;
        }
    </style>
</head>
<body>
    <div id="app">
        <div class="theme-container" :class="isNight ? 'theme-dark' : 'theme-light'">
            <p class="theme-title">{{ isNight ? '夜间模式' : '白天模式' }}</p>
            <hr class="theme-divider">
            <p class="theme-text">{{ isNight ? '夜间模式下页面背景为黑色,文字为浅色,线条为浅色' : '白天模式下页面背景为白色,文字为黑色,线条为深色' }}</p>
            <div class="theme-switch">
                <label>
                    <input type="checkbox" v-model="isNight"> 夜间模式
                </label>
            </div>
        </div>
    </div>
    <script>
        const { createApp } = Vue
        createApp({
            data() {
                return {
                    // 初始化时读取本地存储，若无存储则匹配系统偏好
                    isNight: localStorage.getItem('theme') ? localStorage.getItem('theme') === 'dark' 
                        : window.matchMedia('(prefers-color-scheme: dark)').matches
                }
            },
            watch: {
                // 监听模式变化，同步写入本地存储
                isNight(newVal) {
                    localStorage.setItem('theme', newVal ? 'dark' : 'light')
                }
            }
        }).mount('#app')
    </script>
</body>
